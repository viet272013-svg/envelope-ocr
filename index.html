Hướng dẫn kết nối Google Sheets (Google Apps Script)

  1. Mở file Google Sheets mà bạn muốn lưu dữ liệu.
  2. Ở thanh menu trên cùng, chọn Tiện ích mở rộng (Extensions) > Apps Script.
  3. Xóa toàn bộ mã mặc định và dán đoạn mã bên dưới vào.
  4. Nhấn nút Triển khai (Deploy) > Triển khai dưới dạng ứng dụng web (New deployment).
  5. Chọn **Ai cũng có quyền truy cập (Anyone)** ở mục "Who has access".
  6. Sao chép **Web App URL** thu được và dán vào ô **Google Apps Script Web App URL** ở phần cấu hình ứng dụng web này.
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Thời Gian", "Họ Và Tên", "Ngày Sinh", "Địa Chỉ", "Số Điện Thoại"]);
      sheet.getRange(1, 1, 1, 5).setFontWeight("bold").setBackground("#4F46E5").setFontColor("#FFFFFF");
    }
    
    var data = JSON.parse(e.postData.contents);
    var timestamp = new Date().toLocaleString("vi-VN");
    
    sheet.appendRow([
      timestamp,
      data.name || "",
      data.dob || "",
      data.address || "",
      "'" + (data.phone || "")
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "result": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
